# 尺寸和间距 - Size and Spacing

## 概述

遵循 Jackery 设计系统响应式设计方案，以CSS 根节点`font-size`为基准，使用rem进行换算，实现跨设备的字号大小自适应。

## CSS 变量


```css

/**
 * 尺寸生成Mixin
 * @param {number} @rem-base - rem基准值，用于响应式计算
 * @description 生成一套完整的尺寸变量，支持桌面端和移动端自适应
 */
.generate-sizes(@rem-base) {
  --jds-size-2: max(0, 2 * @rem-base);    /* 2px等效尺寸 */
  --jds-size-4: max(0, 4 * @rem-base);    /* 4px等效尺寸 */
  --jds-size-6: max(0, 6 * @rem-base);    /* 6px等效尺寸 */
  --jds-size-8: max(0, 8 * @rem-base);    /* 8px等效尺寸 */
  --jds-size-10: max(0, 10 * @rem-base);  /* 10px等效尺寸 */
  --jds-size-12: max(0, 12 * @rem-base);  /* 12px等效尺寸 */
  --jds-size-14: max(0, 14 * @rem-base);  /* 14px等效尺寸 */
  --jds-size-16: max(0, 16 * @rem-base);  /* 16px等效尺寸 */
  --jds-size-18: max(0, 18 * @rem-base);  /* 18px等效尺寸 */
  --jds-size-20: max(0, 20 * @rem-base);  /* 20px等效尺寸 */
  --jds-size-22: max(0, 22 * @rem-base);  /* 22px等效尺寸 */
  --jds-size-24: max(0, 24 * @rem-base);  /* 24px等效尺寸 */
  --jds-size-28: max(0, 28 * @rem-base);  /* 28px等效尺寸 */
  --jds-size-32: max(0, 32 * @rem-base);  /* 32px等效尺寸 */
  --jds-size-36: max(0, 36 * @rem-base);  /* 36px等效尺寸 */
  --jds-size-40: max(0, 40 * @rem-base);  /* 40px等效尺寸 */
  --jds-size-48: max(0, 48 * @rem-base);  /* 48px等效尺寸 */
  --jds-size-56: max(0, 56 * @rem-base);  /* 56px等效尺寸 */
  --jds-size-64: max(0, 64 * @rem-base);  /* 64px等效尺寸 */
  --jds-size-80: max(0, 80 * @rem-base);  /* 80px等效尺寸 */
  --jds-size-96: max(0, 96 * @rem-base);  /* 96px等效尺寸 */
}

@media screen and (min-width: 1024px) {
  @rem: (1rem / 16);  /* 桌面端rem基准：1rem = 16px */
  :root {
    /* 生成桌面端尺寸变量 */
    .generate-sizes(@rem);
  }
}

@media screen and (max-width: 1023.98px) {
  @rem: (1rem / 32);  /* 移动端rem基准：1rem = 32px */
  :root {
    /* 生成移动端尺寸变量 */
    .generate-sizes(@rem);
  }
}
```
